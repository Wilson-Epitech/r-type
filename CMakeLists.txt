cmake_minimum_required(VERSION 3.14)
project(r-type)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(VCPKG_ROOT "${CMAKE_SOURCE_DIR}/build/vcpkg")

option(BUILD_SHARED_LIBS ON)

if(NOT DEFINED VCPKG_TARGET_TRIPLET)
    if(UNIX AND NOT APPLE)
        if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64")
            set(VCPKG_TARGET_TRIPLET "x64-linux")
        elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
            set(VCPKG_TARGET_TRIPLET "arm64-linux")
        endif()
    elseif(APPLE)
        if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64")
            set(VCPKG_TARGET_TRIPLET "arm64-osx")
        else()
            set(VCPKG_TARGET_TRIPLET "x64-osx")
        endif()
    elseif(WIN32)
        if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64")
            set(VCPKG_TARGET_TRIPLET "x64-windows")
        elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86")
            set(VCPKG_TARGET_TRIPLET "x86-windows")
        # elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "")
        #     set(VCPKG_TARGET_TRIPLET "arm64-windows")
        endif()
    endif()
endif()

if(UNIX)
    add_compile_options(-Wall -Wextra)
elseif(WIN32)
    add_definitions(-D_WIN32_WINNT=0x0A00)
endif()

if(NOT EXISTS "${VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake")
    message(STATUS "Downloading vcpkg...")
    execute_process(COMMAND git clone https://github.com/microsoft/vcpkg.git ${VCPKG_ROOT})
    if(UNIX)
        execute_process(COMMAND ${VCPKG_ROOT}/bootstrap-vcpkg.sh WORKING_DIRECTORY ${VCPKG_ROOT})
    elseif(WIN32)
        execute_process(COMMAND ${VCPKG_ROOT}/bootstrap-vcpkg.bat WORKING_DIRECTORY ${VCPKG_ROOT})
    endif()
    execute_process(COMMAND ${VCPKG_ROOT}/vcpkg integrate install)
    execute_process(COMMAND ${VCPKG_ROOT}/vcpkg install raylib asio glfw3)
endif()

include_directories("${VCPKG_ROOT}/installed/${VCPKG_TARGET_TRIPLET}/include")
include_directories("${VCPKG_ROOT}/installed/${VCPKG_TARGET_TRIPLET}/include/GLFW")

# message(STATUS "INCLUDE DIRECTORY: ${VCPKG_ROOT}/installed/${VCPKG_TARGET_TRIPLET}/include/asio")

find_package(raylib CONFIG REQUIRED)
find_package(asio CONFIG REQUIRED)
find_package(glfw3 CONFIG REQUIRED)

add_subdirectory(src)
add_subdirectory(src/network/client)
add_subdirectory(src/network/server)
